FROM python:3.11-slim

# Устанавливаем Chromium и зависимости
RUN apt-get update && apt-get install -y \
    chromium \
    chromium-driver \
    libnss3 \
    libxss1 \
    libasound2 \
    libatk-bridge2.0-0 \
    libgtk-3-0 \
    fonts-liberation \
    libgbm1 \
    && rm -rf /var/lib/apt/lists/*

# Hugging Face Spaces требует пользователя с UID 1000
RUN useradd -m -u 1000 user
USER user

# Переменная окружения для пути к браузеру
ENV BROWSER_PATH=/usr/bin/chromium

WORKDIR /home/user/app

# Копируем зависимости и устанавливаем
COPY --chown=user requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Копируем скрипты
COPY --chown=user parser.py .
COPY --chown=user predictor.py .
COPY --chown=user stats.py .

# Запускаем парсер
CMD ["python", "-u", "parser.py"]
